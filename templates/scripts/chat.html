// ============================================ // MULTI-AGENT CHAT // ============================================ async function initializeFiscalChat() { const startChatBtn = document.getElementById('startChatBtn'); const closeChatBtn = document.getElementById('closeChatBtn'); const chatContent = document.getElementById('chat-content'); const recommendationsContent = document.getElementById('recommendations-content'); const chatInput = document.getElementById('chat-input'); const chatSendBtn = document.getElementById('chat-send-btn'); const chatMessages = document.getElementById('chat-messages'); if (!chatContent) return; try { startChatBtn.style.display = 'none'; closeChatBtn.style.display = 'block'; isChatActive = true; recommendationsContent.style.display = 'none'; chatContent.style.display = 'flex'; const formData = getFormData(); currentUserData = formData; chatMessages.innerHTML = '
' + '

' + 'Inicializando expertos fiscales...

'; const calcResponse = await fetch('/api/calculate', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(formData) }); if (!calcResponse.ok) { throw new Error('Error calculando impuestos'); } currentCalculationResult = await calcResponse.json(); const agentsResponse = await fetch('/api/chat/agents', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ calculation_result: currentCalculationResult, user_data: currentUserData, fiscal_year: formData.fiscal_year }) }); if (!agentsResponse.ok) { throw new Error('Error obteniendo agentes'); } chatAgents = await agentsResponse.json(); renderAgentButtons(); chatMessages.innerHTML = `
💬

¡Bienvenido al Chat Fiscal!

Selecciona un experto arriba y haz tu pregunta.

`; conversationHistory = []; selectedAgentId = null; chatInput.value = ''; chatInput.disabled = true; chatSendBtn.disabled = true; await getRecommendationUsage(); } catch (error) { console.error('Error initializing chat:', error); chatMessages.innerHTML = `
⚠️

Error

${error.message}

`; } } function renderAgentButtons() { const agentButtonsContainer = document.getElementById('agent-buttons'); if (!agentButtonsContainer) return; agentButtonsContainer.innerHTML = ''; const agentIcons = { 'Auditor Fiscal': '🔍', 'Planificador Fiscal': '📊', 'Contador Público': '💼', 'Asesor Financiero': '💰', 'Abogado Fiscalista': '⚖️', 'Consultor Empresarial': '🎯' }; chatAgents.forEach(agent => { const button = document.createElement('button'); button.className = 'agent-btn'; button.dataset.agentId = agent.agent_id; const icon = agentIcons[agent.profession] || '👤'; button.innerHTML = ` ${icon} ${agent.name} ${agent.profession} `; button.addEventListener('click', () => selectAgent(agent.agent_id)); agentButtonsContainer.appendChild(button); }); } function selectAgent(agentId) { selectedAgentId = agentId; const chatInput = document.getElementById('chat-input'); const chatSendBtn = document.getElementById('chat-send-btn'); document.querySelectorAll('.agent-btn').forEach(btn => { if (btn.dataset.agentId === agentId) { btn.classList.add('selected'); } else { btn.classList.remove('selected'); } }); if (chatInput && chatSendBtn) { chatInput.disabled = false; chatSendBtn.disabled = false; chatInput.focus(); } } async function sendChatMessage() { const chatInput = document.getElementById('chat-input'); const chatSendBtn = document.getElementById('chat-send-btn'); const chatMessages = document.getElementById('chat-messages'); if (!chatInput || !chatSendBtn || !selectedAgentId) return; const message = chatInput.value.trim(); if (!message) return; try { chatInput.disabled = true; chatSendBtn.disabled = true; document.querySelectorAll('.agent-btn').forEach(btn => btn.disabled = true); chatInput.value = ''; const userMessageDiv = document.createElement('div'); userMessageDiv.className = 'chat-message user'; userMessageDiv.innerHTML = `
👤 Tú
${escapeHtml(message)}
`; chatMessages.appendChild(userMessageDiv); chatMessages.scrollTop = chatMessages.scrollHeight; conversationHistory.push({ role: 'user', content: message }); const selectedAgent = chatAgents.find(a => a.agent_id === selectedAgentId); if (!selectedAgent) throw new Error('Agente no encontrado'); const agentMessageDiv = document.createElement('div'); agentMessageDiv.className = 'chat-message agent'; const agentIcon = { 'Auditor Fiscal': '🔍', 'Planificador Fiscal': '📊', 'Contador Público': '💼', 'Asesor Financiero': '💰', 'Abogado Fiscalista': '⚖️', 'Consultor Empresarial': '🎯' }[selectedAgent.profession] || '👤'; agentMessageDiv.innerHTML = `
${agentIcon} ${selectedAgent.name}
`; chatMessages.appendChild(agentMessageDiv); chatMessages.scrollTop = chatMessages.scrollHeight; chatAbortController = new AbortController(); const response = await fetch('/api/chat/message', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ agent_id: selectedAgentId, message: message, calculation_result: currentCalculationResult, user_data: currentUserData, fiscal_year: currentUserData.fiscal_year, conversation_history: conversationHistory.slice(-10) }), signal: chatAbortController.signal }); if (!response.ok) { throw new Error(`Error ${response.status}: ${response.statusText}`); } const reader = response.body.getReader(); const decoder = new TextDecoder(); let agentResponse = ''; const contentDiv = agentMessageDiv.querySelector('.chat-message-content'); contentDiv.innerHTML = ''; while (true) { const { done, value } = await reader.read(); if (done) break; const chunk = decoder.decode(value); const lines = chunk.split('\n'); for (let line of lines) { if (line.startsWith('data: ')) { try { const data = JSON.parse(line.slice(6)); if (data.type === 'chunk') { agentResponse += data.content; contentDiv.innerHTML = marked.parse(agentResponse); chatMessages.scrollTop = chatMessages.scrollHeight; } else if (data.type === 'complete') { conversationHistory.push({ role: 'agent', content: agentResponse }); } } catch (e) { console.error('Parse error:', e); } } } } chatInput.disabled = false; chatSendBtn.disabled = false; document.querySelectorAll('.agent-btn').forEach(btn => btn.disabled = false); chatInput.focus(); await getRecommendationUsage(); } catch (error) { console.error('Error sending message:', error); const errorDiv = document.createElement('div'); errorDiv.className = 'chat-message agent'; errorDiv.innerHTML = `
⚠️ Error
${error.message.includes('429') ? 'Límite diario alcanzado' : 'Error al enviar mensaje'}
`; chatMessages.appendChild(errorDiv); chatMessages.scrollTop = chatMessages.scrollHeight; chatInput.disabled = false; chatSendBtn.disabled = false; document.querySelectorAll('.agent-btn').forEach(btn => btn.disabled = false); } } function closeFiscalChat() { const startChatBtn = document.getElementById('startChatBtn'); const closeChatBtn = document.getElementById('closeChatBtn'); const chatContent = document.getElementById('chat-content'); if (startChatBtn && closeChatBtn && chatContent) { startChatBtn.style.display = 'block'; closeChatBtn.style.display = 'none'; chatContent.style.display = 'none'; isChatActive = false; if (chatAbortController) { chatAbortController.abort(); chatAbortController = null; } chatAgents = []; selectedAgentId = null; conversationHistory = []; currentCalculationResult = null; currentUserData = null; } } function escapeHtml(text) { const div = document.createElement('div'); div.textContent = text; return div.innerHTML; }

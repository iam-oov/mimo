${filledFields}/${totalFields} campos • ${percentage}% listo`; if (iconElement) iconElement.textContent = '⚡'; } else { progressText.textContent = `¡Listo! ${filledFields}/${totalFields} campos completos`; if (iconElement) iconElement.textContent = '✨'; // Add glow effect when ready const aiCard = progressText.closest('.rounded-xl'); if (aiCard && !aiCard.classList.contains('ai-ready')) { aiCard.classList.add('ai-ready'); } } } // Update usage preview for logged users updateAIUsagePreview(); } // Legacy function - kept for backwards compatibility but now uses sync function updateAIProgressIndicator() { // This function is now called via syncAIPreviewProgress from updateFormProgress // Keeping empty stub for any direct calls } function updateAIUsagePreview() { const usagePreview = document.getElementById('ai-usage-preview'); const usageCounter = document.getElementById('usage-counter'); if (usagePreview && usageCounter) { // Mirror the main usage counter info const counterText = usageCounter.textContent || usageCounter.innerText; if (counterText.includes('Cargando')) { usagePreview.innerHTML = '
Cargando disponibilidad...
'; } else if (counterText.includes('restante') || counterText.includes('análisis')) { // Extract usage info and show compact version const match = counterText.match(/(\d+)\s*(análisis|restante)/); if (match) { const remaining = match[1]; if (parseInt(remaining) === 0) { usagePreview.innerHTML = `
🔥 Sin análisis restantes hoy
Se resetean a medianoche
`; } else { usagePreview.innerHTML = `
✨ ${remaining} análisis disponibles hoy
Ir a recomendaciones ↓ `; } } } } } // Scroll to recommendations section smoothly function scrollToRecommendations() { const recommendationsSection = document.getElementById('recommendations-section'); if (recommendationsSection) { recommendationsSection.scrollIntoView({ behavior: 'smooth', block: 'start' }); } } // Initialize AI progress tracking document.addEventListener('DOMContentLoaded', function() { const form = document.getElementById('taxForm'); if (form) { // Update progress on any input change form.addEventListener('input', updateAIProgressIndicator); form.addEventListener('change', updateAIProgressIndicator); // Initial update setTimeout(updateAIProgressIndicator, 500); } // Update usage preview when main counter updates const usageCounter = document.getElementById('usage-counter'); if (usageCounter) { const observer = new MutationObserver(updateAIUsagePreview); observer.observe(usageCounter, { childList: true, characterData: true, subtree: true }); } });

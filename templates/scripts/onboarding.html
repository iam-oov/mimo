`; document.body.appendChild(welcomeOverlay); // Event listeners document.getElementById('tour-example-btn').addEventListener('click', () => { welcomeOverlay.remove(); loadExampleData(); completeTour(); }); document.getElementById('tour-start-btn').addEventListener('click', () => { welcomeOverlay.remove(); showQuickTips(); completeTour(); }); document.getElementById('tour-skip-btn').addEventListener('click', () => { welcomeOverlay.remove(); completeTour(); }); }, 1000); // Show after 1 second } // Show quick tips highlighting key features function showQuickTips() { const tips = [ { element: '#form-progress-container', message: '👆 Completa el formulario y ve tu progreso en tiempo real', position: 'bottom' }, { element: '#kpi-balance', message: '💰 Aquí verás tu saldo estimado a favor o a pagar', position: 'left' }, { element: '[href="/auth/google"]', message: '🚀 Desbloquea análisis AI personalizados con tu cuenta Google', position: 'left' } ]; let currentTip = 0; function showTip(index) { if (index >= tips.length) { return; } const tip = tips[index]; const element = document.querySelector(tip.element); if (!element) { showTip(index + 1); return; } // Create spotlight const spotlight = document.createElement('div'); spotlight.className = 'fixed inset-0 z-[90] pointer-events-none'; spotlight.style.background = 'rgba(0, 0, 0, 0.7)'; // Create tooltip const tooltip = document.createElement('div'); tooltip.className = 'fixed z-[95] bg-slate-900 border border-amber-500/50 rounded-lg p-4 shadow-xl max-w-xs animate-scale-in'; tooltip.innerHTML = `

${tip.message}

`; // Position tooltip const rect = element.getBoundingClientRect(); if (tip.position === 'bottom') { tooltip.style.top = `${rect.bottom + 10}px`; tooltip.style.left = `${rect.left}px`; } else if (tip.position === 'left') { tooltip.style.top = `${rect.top}px`; tooltip.style.right = `${window.innerWidth - rect.left + 10}px`; } document.body.appendChild(spotlight); document.body.appendChild(tooltip); // Highlight element element.classList.add('ring-2', 'ring-amber-500', 'ring-offset-2', 'ring-offset-slate-900', 'relative', 'z-[91]'); element.scrollIntoView({ behavior: 'smooth', block: 'center' }); // Event listeners tooltip.querySelector('.next-tip-btn').addEventListener('click', () => { spotlight.remove(); tooltip.remove(); element.classList.remove('ring-2', 'ring-amber-500', 'ring-offset-2', 'ring-offset-slate-900', 'z-[91]'); showTip(index + 1); }); tooltip.querySelector('.skip-tour-btn').addEventListener('click', () => { spotlight.remove(); tooltip.remove(); element.classList.remove('ring-2', 'ring-amber-500', 'ring-offset-2', 'ring-offset-slate-900', 'z-[91]'); }); } showTip(0); } // Mark tour as completed function completeTour() { localStorage.setItem('mimo-tour-completed', 'true'); } // Initialize onboarding function initOnboarding() { // Add event listener to "Ver Ejemplo" button const loadExampleBtn = document.getElementById('load-example-btn'); if (loadExampleBtn) { loadExampleBtn.addEventListener('click', loadExampleData); } // Show tour for first-time users (after page load) if (!localStorage.getItem('mimo-tour-completed')) { showOnboardingTour(); } }
